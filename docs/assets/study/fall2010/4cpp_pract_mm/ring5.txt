using namespace std;
//Раскинулось поле по модулю пять...
class Ring5{
	friend ostream& operator<<(ostream& os, const Ring5& R);
	friend istream& operator>>(istream& is, Ring5& R);
	friend const Ring5 operator-(const Ring5& r1, const Ring5& r2);
private:
	static const int mod = 5;
	int val;
public:
	Ring5(int val = 0) : val((mod + val % mod) % mod){
		;
	}

	//Приведение типа. ВНИМАНИЕ! Здесь не надо указывать возвращаемый тип
	operator int()const{
		return val;
	}

	//Оператор присваивания. Теперь можно присваивать b = a = 4;
	Ring5& operator=(int val){
		this->val = (mod + val % mod) % mod;
		return *this;
	}

	Ring5 operator+(const Ring5& R) const{
		return Ring5(this->val + R.val);
	}

	//Можно было перегрузить (-) как (+), но мы попробуем иначе=)
	/*
	const Ring5 operator-(const Ring5& R) const{
		return Ring5(this->val - R.val);
	}
	*/



	Ring5 operator-() const{		
		return Ring5(-val);
	}

	Ring5 operator~() const{	
		//PHI(5) - 1 == (5 - 1) - 1 == 3  (=>) a^(-1) === a^3 mod 5 
		return Ring5(val*val*val);
	}

	Ring5& operator +=(const Ring5& R){
		(this->val += R.val) %= mod;
	}

	//Prefix form (++a)
	Ring5& operator++(){
		(++this->val) %= mod; 
		return *this;
	}

	//Postfix form (a++)
	//Постфиксная форма определяется именно так --- через мнимый целочисленный параметр
	//тут const обязательно писать, иначе можно получить ошибку. i++++ --- увеличит значение на 1, а не на 2
	const Ring5 operator++(int){
		Ring5 prev(*this);
		(++this->val) %= mod;
		return prev;
	}
};


//Операторы потока не перегрузить иначе --- у нас ведь нет доступа к классам iostream
ostream& operator<<(ostream& os, const Ring5& R){
	return os << R.val << "(" << R.mod << ")";
}

istream& operator>>(istream& is, Ring5& R){
	int i;
	is >> i;
	R = i;
	return is;
}


//Симметричные операторы кажется логичным перегружать именно так
const Ring5 operator-(const Ring5& r1, const Ring5& r2){
	return Ring5(r1.val - r2.val);
}

//Функциональный объект =)
class Function{
public:
	virtual double operator()(double x) const = 0;
};

class ParabolicFunction : public Function{
double a, b, c;
public:
	ParabolicFunction(double a = .0, double b = .0, double c =.0) : a(a), b(b), c(c){
		;
	}

	double operator[](int i) const{
		switch(i){
			case 0: return a;
			case 1: return b;
			default: return c;
		}
	}
	
	double operator()(double x) const{
		return a*x + b*x + c;
	}

	double operator()(double x, double y) const{
		return a*x*x + b*x + c - y;
	}

};


//Так работаем с []
class Array{
	double *p;
	size_t size;
public:
	Array(size_t size = 0) : size(size){
		p = new double[size];
	}

	~Array(){
		delete [] p;
	}

	Array(const Array& A) : size(A.size){
		this->p = new double[size];
		memcpy(this->p, A.p, size*sizeof(double));
	}

	Array& operator=(const Array& A){
		delete [] p;
		this->size = A.size;
		this->p = new double[size];
		memcpy(this->p, A.p, size*sizeof(double));
		return *this;
	}

	double& operator[](size_t inx){
		return p[inx];
	}

	const double& operator[](size_t inx) const {
		return p[inx];
	}
};
